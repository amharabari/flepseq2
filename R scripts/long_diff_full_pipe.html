my_pRepare <- function(x) { 
  
  x$variable <- factor(x$Genotype, levels=c("Col0", "xrn4", "xrn4_dne1"))
  x$Percent_co <- ifelse(x$Percent==0, 0.1/x$total*100, x$Percent)
  
  x$Long <- as.numeric(x$Percent_co)
  x$NonLong <- as.numeric(100 - x$Percent_co)
  
  x$Rep_V <- factor(x$Rep, levels=c("Rep1", "Rep2", "Rep3"))
  x$Feature.name_V <- factor(x$Feature.name)
  return(x)
}

# ---- Fitlering to have the same Feature.name in all Rep x Genotype combinations -----
filter_A_for_all_BC <- function(df) {
  # Extraire toutes les combinaisons distinctes de Genotype et Rep
  all_BC <- df %>%
    distinct(Genotype, Rep)
  
  # Pour chaque Feature.name, compter combien de combinaisons (B,C) il couvre
  valid_A <- df %>%
    distinct(Genotype, Rep, Feature.name) %>%
    group_by(Feature.name) %>%
    summarise(n_bc = n()) %>%
    ungroup()
  
  # Nombre total de combinaisons Genotype, Rep dans la table
  total_bc <- nrow(all_BC)
  
  # Garder les Feature.name qui ont toutes les combinaisons Genotype, Rep
  valid_A <- valid_A %>%
    filter(n_bc == total_bc) %>%
    pull(Feature.name)
  
  # Filtrer le df original pour garder uniquement les lignes avec ces A valides
  df %>% filter(Feature.name %in% valid_A)
}

table_glht <- function(x) {
  pq <- summary(x)$test
  mtests <- cbind(pq$coefficients, pq$sigma, pq$tstat, pq$pvalues)
  error <- attr(pq$pvalues, "error")
  colnames(mtests) <- c("Estimate", "Std.Error", "z.value", "p.value")
  return(mtests)
}

  print("Summary of the model")
  print(summary(model))

  Test <- anova(model, test = "F")
  print("ANOVA: Results of the F test")
  print(Test)

                family = quasibinomial,

    Genotype = names(letters_vec),

               data = Data)
  print("Summary of the model")
  print(summary(model))

  print(Test)

  print("Confidence interval")
  model2 <- glm(cbind(Long, NonLong) ~ variable - 1,
                family = quasibinomial,
                data = Data)
  print(plogis(confint(model2, trace = TRUE))) # CIs on log-odds and then transform to probabilities

  tuk_raw <- glht(model, mcp(variable = "Tukey"))
  tuk <- summary(tuk_raw)
  print("Tukey test:")
  print(tuk)
  
  cld_res <- cld(tuk)

  letters_vec <- cld_res$mcletters$Letters
  letters_df <- data.frame(
    Genotype = names(letters_vec),
    group    = as.character(letters_vec),
    stringsAsFactors = FALSE)
  
  # Odds ratio = probability of certain amount of transcripts getting Longer tails for each genotype
  # Odds ratios and CI on the contrasts
  coef  <- tuk$test$coef          # log-odds differences
  se    <- tuk$test$sigma         # standard errors
  pval  <- tuk$test$pvalues       # adjusted p-values
  
  ORs   <- exp(coef)
  tuk_ci <- confint(tuk_raw)$confint
  lower  <- exp(tuk_ci[, "lwr"])
  upper  <- exp(tuk_ci[, "upr"])
  
    # Counts per genotype
  counts_df <- as.data.frame(table(Data$variable))
  colnames(counts_df) <- c("Genotype", "n_obs")

  # Merge letters and counts by Genotype
  letters_counts <- merge(letters_df, counts_df, by = "Genotype", all.x = TRUE)

  OR_table <- data.frame(
  Contrast = names(coef),
  OR       = round(ORs,   3),
  CI_low   = round(lower, 3),
  CI_high  = round(upper, 3),
  p_adj    = signif(pval, 3),  # 3 significant digits, not 3 decimals
  row.names = NULL,
  stringsAsFactors = FALSE
)
  # Prepare a combined export table:
  # Part 1: ORs (contrast-level)
  OR_export <- OR_table
  OR_export$type <- "contrast"

  # Part 2: letters + counts (genotype-level)
  letters_export <- letters_counts
  letters_export$Contrast <- letters_export$Genotype
  letters_export$OR       <- NA_real_
  letters_export$CI_low   <- NA_real_
  letters_export$CI_high  <- NA_real_
  letters_export$p_adj    <- NA_real_
  letters_export$type     <- "genotype"

  OR_export$Genotype <- NA_character_
  OR_export$group    <- NA_character_
  OR_export$n_obs    <- NA_integer_

  OR_export  <- OR_export[,  c("type","Contrast","Genotype","group","n_obs","OR","CI_low","CI_high","p_adj")]
  letters_export <- letters_export[, c("type","Contrast","Genotype","group","n_obs","OR","CI_low","CI_high","p_adj")]

  combined_table <- rbind(OR_export, letters_export)

  print("Combined OR + letters + counts table:")
  print(combined_table)
  
  return(list(
    letters        = letters_df,
    OR_table       = OR_table,
    combined_table = combined_table
  ))
}

my_glm_test <- function(x) {
  
  Result_table_exp <- data.frame(Estimate=as.numeric(), Std.Error=as.numeric(), z.value=as.numeric(), p.value=as.numeric(), comparaison=as.character(), Feature.name=as.character(), padj=as.character(), Condition=as.character())
  
  for (t in unique(x$Condition))  {
    cat("Condition:", t, "\n")
    
    Result_table <- data.frame(Estimate=as.numeric(), Std.Error=as.numeric(), z.value=as.numeric(), p.value=as.numeric(), comparaison=as.character(), Feature.name=as.character())
    Selec_condition <- x[x$Condition==t,]
    Selec_condition <- filter_A_for_all_BC(Selec_condition)
    
    
    for (i in unique(Selec_condition$Feature.name)) {
     # print(i)
      PermRNA_percent_50_tailed_U_gene <- Selec_condition[Selec_condition$Feature.name==i, ]
      if (nrow(PermRNA_percent_50_tailed_U_gene) <= 2) {next}
      tryCatch({     
        Uri1 <- glm(cbind(Long , NonLong   ) ~ variable, family=quasibinomial, data=PermRNA_percent_50_tailed_U_gene)
        glht_obj <- summary(glht(Uri1 , mcp(variable="Tukey")))
        Tab <- as.data.frame(table_glht(glht_obj))
        Tab$comparaison <- row.names(Tab)
        Tab$Feature.name <- i
        rownames(Tab) <- NULL
        Result_table <- rbind(Result_table, Tab)
      }, error = function(e) {
        message("Skipping ", i, ": ", e$message)
      })
    }
    
    
    Result_table$padj <- p.adjust(Result_table$p.value, "BH")
    Result_table$Condition <- t
    Result_table_exp <- rbind(Result_table_exp, Result_table)
    
  }
  
  return(Result_table_exp)
}
